\begin{frame}{1. Продукт}
\begin{block}{Одной фразой}
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item \textbf{Продукт:} веб-приложение для разметки действий на видео
  \item \textbf{Пользователи:} CV/ML инженеры, CV-эксперты и разметчики
  \item \textbf{Решение:} авточерновик разметки $\rightarrow$ проверка человеком $\rightarrow$ экспорт JSON/CSV
\end{itemize}
\end{block}

\vspace{8pt}
\begin{columns}[T,onlytextwidth]
\column{0.55\textwidth}
\begin{itemize}
  \setlength{\itemsep}{8pt}
  \item \pill{BrandBlue}{Input} видео (MP4/MOV, $\le 30$ сек)
  \item \pill{BrandOrange}{AI} шаги: \textbf{action, object, keyframe}
  \item \pill{BrandGreen}{HITL} правка: границы, классы, кадры
\end{itemize}

\column{0.45\textwidth}
\begin{alertblock}{Почему это важно}
\begin{itemize}
  \setlength{\itemsep}{8pt}
  \item Ручная разметка: \textbf{15--30 мин} / ролик
  \item 1000 роликов: \textbf{сотни часов}
\end{itemize}
\end{alertblock}
\end{columns}
\end{frame}

\begin{frame}{3. Демонстрация решения}
\begin{block}{Фокус: один ключевой сценарий end-to-end}
\begin{enumerate}
  \setlength{\itemsep}{4pt}
  \item \pill{BrandBlue}{Upload} загрузить видео
  \item \pill{BrandOrange}{Run} запустить pipeline на бэкенде
  \item \pill{BrandGray}{Review} открыть таймлайн шагов (confidence, keyframe)
  \item \pill{BrandGreen}{Fix} исправить \textbf{одну} ошибку
